\subsection{Phần mềm}

\begin{itemize}
  \item \textbf{Ngôn ngữ:} Python 3.10+.
  \item \textbf{Học sâu và phát hiện:} PyTorch (bản CUDA 12.x), Ultralytics (YOLO26, YOLO11, RT-DETR, ByteTrack).
  \item \textbf{Nhận dạng ký tự:} fast-plate-ocr, mô hình CRNN tự cài đặt bằng PyTorch; EasyOCR, PaddleOCR làm baseline.

  \item \textbf{Triển khai:} ONNX Runtime, FastAPI, SQLite, Docker, Cloudflare Tunnel.
  \item \textbf{Quản lý mã nguồn và gán nhãn:} Git/GitHub, Roboflow hoặc CVAT.
\end{itemize}

\section{Kế hoạch triển khai mã nguồn theo từng giai đoạn}

\begin{enumerate}
  \item \textbf{Giai đoạn 1 -- Dữ liệu:}
  \begin{itemize}
    \item Script đọc nhãn từ tên tệp VNLP, kiểm tra định dạng, chuẩn hóa chuỗi.
    \item Sinh bộ detection (YOLO, 2 lớp) và bộ OCR (ảnh biển cắt), chia theo biển số.
    \item Thu thập video và bộ dữ liệu công khai, trích khung hình, lọc ảnh trùng với VNLP, gán nhãn làm tập kiểm thử thực tế.
    \item Mã nguồn xử lý dữ liệu được tổ chức thành các Jupyter Notebook: phân tích và làm sạch nhãn, kiểm tra ảnh, tạo bộ dữ liệu, xử lý dữ liệu ngoài.
  \end{itemize}

  \item \textbf{Giai đoạn 2 -- Phát hiện:}
  \begin{itemize}
    \item Huấn luyện YOLO26, YOLO11, RT-DETR trên Kaggle với cùng cấu hình.
    \item Đánh giá trên tập kiểm thử VNLP và tập thực tế; xuất mô hình được chọn sang ONNX.
  \end{itemize}

  \item \textbf{Giai đoạn 3 -- Nhận dạng ký tự:}
  \begin{itemize}
    \item Đo baseline EasyOCR, PaddleOCR trên ảnh biển cắt từ nhãn.
    \item Huấn luyện CRNN-CTC và tinh chỉnh fast-plate-ocr trên máy cá nhân.
    \item Thí nghiệm có và không có tiền xử lý ảnh.
  \end{itemize}

  \item \textbf{Giai đoạn 4 -- Video:}
  \begin{itemize}
    \item Tích hợp ByteTrack, cài đặt bỏ phiếu đa khung hình và luật định dạng biển.
    \item Chương trình chạy thống nhất cho ảnh, video và webcam.
  \end{itemize}

  \item \textbf{Giai đoạn 5 -- Tích hợp và triển khai:}
  \begin{itemize}
    \item Máy chủ FastAPI và giao diện web; lưu lịch sử bằng SQLite.
    \item Đo tốc độ thời gian thực trên máy cá nhân; đóng gói Docker; triển khai qua Cloudflare Tunnel.
  \end{itemize}
\end{enumerate}

\section{Kịch bản thử nghiệm}

\begin{itemize}
  \item \textbf{Trong miền (VNLP):} tập kiểm thử VNLP chia theo biển số, báo cáo riêng ba nhóm và riêng mã 36 với mã tỉnh khác.
  \item \textbf{Ngoài miền (tập thực tế):}
  \begin{itemize}
    \item \textit{Ban ngày, góc gần trực diện:} kịch bản thuận lợi.
    \item \textit{Ban đêm, ngược sáng:} đèn pha gây lóa, độ tương phản thấp.
    \item \textit{Xe chuyển động nhanh:} mờ chuyển động.
    \item \textit{Biển xa, nhỏ; biển 2 dòng; biển bẩn, che khuất một phần.}
  \end{itemize}
\end{itemize}

\section{Kịch bản so sánh}

\begin{itemize}
  \item \textbf{So sánh mô hình phát hiện:} YOLO26, YOLO11, RT-DETR theo mAP, tốc độ và kết quả đọc toàn biển khi ghép với bộ nhận dạng.
  \item \textbf{So sánh phương pháp nhận dạng:} EasyOCR, PaddleOCR, CRNN-CTC, fast-plate-ocr theo tỷ lệ đọc đúng toàn biển, CER và tốc độ.
  \item \textbf{Ảnh hưởng của tiền xử lý:} cùng một bộ nhận dạng, có và không có CLAHE, nắn phối cảnh.
  \item \textbf{Ảnh hưởng của xử lý video:} đọc từng khung hình đơn lẻ so với bỏ phiếu đa khung hình kết hợp luật định dạng.
  \item \textbf{Khoảng cách miền dữ liệu:} cùng một mô hình trên tập VNLP và trên tập thực tế.
\end{itemize}

\section{Kết quả thực nghiệm}

\textit{Mục này được cập nhật sau khi hoàn thành thực nghiệm.} Kết quả sẽ được trình bày theo các bảng: so sánh detector, so sánh phương pháp nhận dạng, kết quả toàn hệ thống trên hai tập kiểm thử và tốc độ xử lý trên máy cá nhân.

\section{Phương pháp phân tích lỗi}

Các trường hợp sai được phân loại theo nguyên nhân:
\begin{itemize}
  \item \textbf{Lỗi phát hiện:} biển quá nhỏ, bị che khuất hoặc hộp lệch cắt mất ký tự.
  \item \textbf{Lỗi chất lượng ảnh:} lóa sáng ban đêm, mờ chuyển động.
  \item \textbf{Lỗi nhận dạng:} nhầm các cặp ký tự gần giống (8/B, 0/D/O, 1/I, 5/S); đảo thứ tự dòng ở biển 2 dòng.
  \item \textbf{Lỗi do lệch dữ liệu:} sai tập trung ở biển ngoài mã 36 hoặc ở cảnh không có trong VNLP.
\end{itemize}

Nhóm lập ma trận nhầm lẫn ký tự và thống kê lỗi theo từng nhóm để đề xuất cải tiến tương ứng.
